name: R3 Holm checkpoint candidate
on:
  pull_request:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  candidate:
    runs-on: ubuntu-24.04
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "24.19.0"
          cache: pnpm
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12.14"
      - run: pnpm install --frozen-lockfile
      - name: Candidate semantics and old-bundle regression
        run: |
          export NOMUE_TEST_OUTPUT="$RUNNER_TEMP/holm-observations"
          export NOMUE_EXPERIMENT_PYTHON="$(command -v python3)"
          node --import tsx governance/drafts/release-3-preparation/holm-repaired-candidate-20260911/test_envelope.mjs
          node --import tsx governance/drafts/release-3-preparation/holm-repaired-candidate-20260911/test_public.mjs
          node governance/drafts/release-3-preparation/holm-repaired-candidate-20260911/test_budget.mjs
      - run: node governance/drafts/release-3-preparation/r3-review-repair-20260911/audit.mjs
      - run: python3 governance/drafts/release-3-preparation/r3-followup-review-20260911/check_evidence.py
      - run: git diff --exit-code
      - name: Retain candidate observations
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: r3-holm-checkpoint-candidate
          path: |
            ${{ runner.temp }}/holm-observations/RESULTS.json
            ${{ runner.temp }}/holm-observations/PUBLIC-RESULTS.json
            ${{ runner.temp }}/holm-observations/BUDGET-RESULTS.json
